cmake_minimum_required(VERSION 3.16)
project(routes_extract CXX C)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(stormlib GIT_REPOSITORY https://github.com/ladislav-zezula/StormLib.git GIT_TAG master GIT_SHALLOW TRUE)
set(STORM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(stormlib)
add_executable(routes_extract routes_extract.cpp)
target_link_libraries(routes_extract storm)
if(MSVC)
  set_property(TARGET routes_extract PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  set_property(TARGET storm PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
